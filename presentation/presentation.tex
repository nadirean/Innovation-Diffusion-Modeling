\section*{Zadanie 1 - Wprowadzenie, SOTA i sformułowanie modelu}

\begin{frame}{Wprowadzenie}
\begin{itemize}
  \item \textbf{Dyfuzja innowacji} – proces upowszechniania się nowych produktów lub technologii w społeczeństwie, przechodzący kolejne fazy adopcji kolejnych grup odbiorców \cite{wiki:dyfuzja}.
  \item Proces adopcji przebiega początkowo wolno, następnie nabiera tempa, często zgodnie z charakterystyczną krzywą S (funkcja logistyczna) \cite{bass:1969}.
  \item Wzrost popytu na innowację wspierają zarówno innowatorzy (early adopters), jak i mechanizm naśladownictwa („word-of-mouth”).
  \item Modele matematyczne umożliwiają kwantyfikację dynamiki popytu i adopcji (prognozowanie rozmiaru rynku) \cite{hashemi2011}.
\end{itemize}
\end{frame}

\begin{frame}{Klasyczne modele matematyczne dyfuzji}
\begin{itemize}
  \item \textbf{Model Bassa} (1969) – równanie różniczkowe opisujące tempo adopcji nowego produktu. Zakłada dwie kategorie nabywców: \emph{innowatorów} (efekt zewnętrzny) i \emph{naśladowców} (efekt imitacji). Równanie: $\frac{dF}{dt}=p(1-F)+qF(1-F)$, gdzie $F(t)$ to skumulowany udział adopcji, $p$ – współczynnik innowacji, $q$ – współczynnik imitacji \cite{bass:1969}. Rozwiązanie daje S-kształtną krzywą adopcji: początkowo szybki wzrost, potem nasycenie.
  \item \textbf{Równanie Fishera–Kolmogorowa–Petrowskiego–Piskunowa (F-KPP)} – reakcyjno-dyfuzyjne równanie PDE: $u_t = D \Delta u + r\,u(1-u)$ \cite{fisher1937}. Proponowane w genetyce populacji (Fisher 1937) do opisania fali rozprzestrzeniania korzystnego genu. W ekonomii modeluje propagację innowacji jako falę nasycenia (przejście od stanu bez adopcji do pełnej adopcji).
  \item \textbf{Przebieg krzywej adopcji} – obydwa powyższe modele generują krzywę typu logistycznego (S-kształtną): szybka faza początkowa (wzrost adopcji) i faza wygaszania przy nasyceniu rynku \cite{bass:1969,fisher1937}.
\end{itemize}
\end{frame}

\begin{frame}{Uogólnienie do modeli przestrzenno-czasowych (PDE)}
\begin{itemize}
  \item Dodanie wymiaru przestrzennego do modelu Bassa prowadzi do \emph{równania cząstkowego różniczkowego}: 
  $$\frac{\partial F(x,t)}{\partial t} = D \Delta F + p\,(1-F) + q\,F(1-F)\,,
  $$
  gdzie $D$ to współczynnik dyfuzji (intensywność rozprzestrzeniania informacji w przestrzeni) \cite{dubinina2023}.
  \item Takie modele reakcji–dyfuzji uwzględniają mechanizmy lokalnej interakcji (zależność od odległości przestrzennej) oraz pierwiastek losowy/adopcyjny w zachowaniach konsumentów.
  \item Rozwiązania przyjmują formę propagujących się fal innowacji (autowal). Ich prędkość opisuje tempo rozwoju popytu w przestrzeni – np. Dubinina (2023) analitycznie wyprowadza falę podróżującą, określając czas osiągnięcia danego poziomu technologii \cite{dubinina2023}.
  \item W praktyce modele PDE pozwalają analizować klastrowe rozprzestrzenianie się innowacji (np. koncentrujące się „fale adopcji” wokół centrów innowacyjności) oraz prognozować przestrzenne rozkłady popytu.
\end{itemize}
\end{frame}

\begin{frame}{Sformułowanie konkretnego modelu}
\begin{itemize}
  \item Rozważmy zmienną $u(x,t)\in[0,1]$ – skumulowany udział rynku opanowany przez innowację w punkcie $x$ i czasie $t$. Proponowany model:
  $$\frac{\partial u}{\partial t} = D\,\Delta u + (p + q\,u)\,(1-u)\,. $$
  \item Interpretacja parametrów: $p$ – stopa innowacji (zewnętrzna, np. reklama), $q$ – współczynnik imitacji (wewnętrzny, efekt „liczenia się z innymi”), $D$ – współczynnik dyfuzji (szybkość propagacji informacji w przestrzeni).
  \item Warunki początkowe i brzegowe: np. $u(x,0)=u_0(x)$ z niewielką liczbą pionierów w centrum, \; warunki Neumanna ($\nabla u\cdot n=0$) na granicach.
  \item Model przewiduje przyspieszony wzrost adopcji w początkowej fazie i asymptotyczne nasycenie rynku ($u\to1$). Rozwiązania (np. numeryczne lub w przybliżeniu falowe) pozwalają wyznaczyć czas rozprzestrzeniania innowacji i tempo wzrostu popytu \cite{hashemi2011}.
\end{itemize}
\end{frame}

\begin{frame}{Podsumowanie i zastosowania}
\begin{itemize}
    \item Modele przestrzenno-czasowe umożliwiają uwzględnienie efektów geograficznych: jak bliskość i sieci społeczne przyspieszają dyfuzję innowacji \cite{hashemi2011}.
  \item Zastosowania: prognozowanie rozkładu adopcji w regionach, identyfikacja klastrów innowacji, planowanie strategii marketingowych z uwzględnieniem lokalizacji (marketing geograficzny).
    \item Kalibracja na danych empirycznych (np. historia adopcji technologii mobilnej) umożliwia przewidywanie tempa wzrostu popytu i oceny opóźnień geograficznych (patrz Dubinina 2023) \cite{dubinina2023}.
    \item Dalszy rozwój modeli może uwzględniać heterogeniczność populacji, zależność od cen czy czynniki sieciowe, co można wprowadzić jako dodatkowe składniki PDE (np. sprzężenia zwrotne, nieliniowe czynniki) \cite{hashemi2011}.
\end{itemize}
\end{frame}

\section*{Zadanie 2 - Porównanie solverów dla modelu przestrzenno-czasowego}

\begin{frame}{Wprowadzenie}
\begin{itemize}
  \item Rozważamy model reakcyjno-dyfuzyjny dla udziału adopcji $u(x,t)\in[0,1]$:
  $$\frac{\partial u}{\partial t} = D\,\frac{\partial^2 u}{\partial x^2} + (p + q\,u)\,(1-u)\,. $$ 
  \item Parametry: $D$ (dyfuzja), $p$ (innowacja), $q$ (imitacja). Warunki Neumanna na brzegach oraz lokalne warunki początkowe (pionierzy w centrum).
  \item Celem: porównać wydajność i stabilność rozwiązania dla różnych solverów oraz zidentyfikować bezpieczne przedziały parametrów.
\end{itemize}
\end{frame}

\begin{frame}{Metoda numeryczna — solvery testowane}
\begin{itemize}
  \item explicit: jawny Euler w czasie + centralna druga różnica w przestrzeni (najtańszy krok, ograniczony warunkiem CFL).
  \item semi-implicit: dyfuzja implicit (rozwiązanie układu liniowego), reakcja jawnie.
  \item crank-nicolson (CN): CN dla dyfuzji (2. rząd dokładności), reakcja jawnie; większy koszt na krok.
\end{itemize}
\end{frame}

\begin{frame}{Plan eksperymentu}
\begin{itemize}
  \item Siatki w przestrzeni: $nx\in\{50,100,200\}$.
  \item Kroki czasowe testowane: $\Delta t\in\{5\cdot10^{-5},\,10^{-4}\}$, całkowity czas symulacji: $T=0.05$ (szybkie testy). 
  \item Zestaw parametrów (D,p,q): (0.01,0.01,1.0), (0.1,0.001,5.0), (1.0,0.1,10.0).
  \item Mierzone metryki: czas wykonania (s), wykresy profilu $u(x)$, kontrola czy $u\in[0,1]$ i czy występują NaN/Inf.
\end{itemize}
\end{frame}

Wyniki z przebiegu eksperymentu (nagłówek: średni czas wyk. dla wszystkich kombinacji testowych):

  \item semi-implicit — średni czas: $\approx 0.0329\,$s (min $\approx0.0164\,$s, max $\approx0.0568\,$s)
  \item crank-nicolson — średni czas: $\approx 0.0350\,$s (min $\approx0.0178\,$s, max $\approx0.0596\,$s)

\begin{frame}{Wnioski}
\begin{itemize}
  \item (1) Czas obliczeń: dla krótkich testów jawny solver był najszybszy ze względu na niski koszt pojedynczego kroku; schematy rozwiązujące układy liniowe były wolniejsze.
  \item (2) Stabilność: jawny (explicit) solver wymaga mniejszego kroku czasowego $\Delta t$ przy wzroście $D$ (CFL \(\sim\) $\mathrm{dx}^2/D$); schematy semi-implicit i CN są bardziej odporne, ale droższe.
  \item (3) Parametry $p,q$: większe $q$ przyspieszają nasycenie i mogą prowadzić do szybkich wzrostów lokalnych; zalecamy testy kalibracyjne na danych empirycznych.
  \item (4) Zakresy parametryczne (przykładowo): dla stabilnych symulacji przy $nx=100$ i $dt=10^{-4}$ przykładowe granice to: $D\lesssim0.1$ (dla explicit), $p\in[10^{-4},0.1]$, $q\in[0.1,10]$ — jednak ostateczne granice wymagają szczegółowego testu stabilności i dokładności.
\end{itemize}
\end{frame}

\section*{Zadanie 3 — Model surogatowy ODE dla dyfuzji innowacji}

\begin{frame}{Wprowadzenie do modelu surogatowego}
\begin{itemize}
  \item \textbf{Cel:} Uproszczenie modelu PDE do modelu ODE poprzez eliminację zmiennej przestrzennej. Pozwala to na znacznie szybsze symulacje.
  \item \textbf{Metoda:} Uśrednienie równania PDE po całej dziedzinie przestrzennej $x \in [0, 1]$.
  $$ \frac{d}{dt} \int_0^1 u(x,t) dx = \int_0^1 D u_{xx} dx + \int_0^1 (p+qu)(1-u) dx $$
  \item Przy warunkach brzegowych Neumanna, człon dyfuzyjny się zeruje: $\int_0^1 D u_{xx} dx = D[u_x]_0^1 = 0$.
  \item Definiując $U(t) = \mathbb{E}[u(x,t)] = \int_0^1 u(x,t) dx$ i stosując przybliżenie $\mathbb{E}[R(u)] \approx R(\mathbb{E}[u])$, otrzymujemy model surogatowy ODE:
  $$ \frac{dU}{dt} = (p + qU)(1-U) $$
\end{itemize}
\end{frame}

\begin{frame}{Sieć neuronowa jako model surogatowy}
\begin{itemize}
    \item \textbf{Cel:} Stworzenie uniwersalnego modelu surogatowego, który generalizuje dynamikę dla różnych parametrów $D, p, q$.
    \item \textbf{Dane uczące:} Zbiór symulacji PDE dla losowych kombinacji parametrów (uczenie na wielu przebiegach).
    \item \textbf{Architektura:} Sieć MLP z wejściem 4-wymiarowym $(t, D, p, q)$, dwiema warstwami ukrytymi i wyjściem $U(t)$.
    \item \textbf{Wynik:} Sieć nauczyła się odwzorowywać wpływ parametrów na krzywą adopcji, co pozwala na natychmiastową predykcję dla nowych, nieznanych wcześniej parametrów.
\end{itemize}
\end{frame}

\begin{frame}{Porównanie wszystkich trzech modeli}
\begin{figure}
  \centering
  \includegraphics[width=0.7\textwidth]{comparison_plot.png}
  \caption{Porównanie średniej z modelu PDE, modelu surogatowego ODE oraz predykcji sieci neuronowej.}
\end{figure}
\begin{itemize}
    \item Parametry testowe: $D=0.02, p=0.079, q=6.68$.
    \item Czas inferencji dla modelu ODE: $\approx 1.29$ ms.
    \item Czas inferencji dla modelu NN: $\approx 0.87$ ms.
\end{itemize}
\end{frame}

\begin{frame}{Wnioski końcowe}
\begin{itemize}
    \item Modele surogatowe (zarówno analityczne ODE, jak i sieci neuronowe) są potężnym narzędziem do przyspieszania symulacji złożonych systemów PDE.
    \item Model ODE, mimo swojej prostoty, dostarcza dobrego przybliżenia ogólnej dynamiki systemu.
    \item Sieć neuronowa wytrenowana na zbiorze symulacji potrafi generalizować i przewidywać dynamikę dla nowych parametrów z bardzo wysoką dokładnością i w czasie poniżej 1 ms.
    \item Wybór odpowiedniego modelu surogatowego zależy od wymagań dotyczących dokładności, dostępności danych i potrzeby interpretowalności wyników.
\end{itemize}
\end{frame}

\section{Zadanie 4: Analiza Wrażliwości}

\begin{frame}{Cel i Metodyka Analizy Wrażliwości}
    \begin{itemize}
        \item \textbf{Cel:} Zidentyfikowanie, które parametry wejściowe ($D, p, q$) mają największy wpływ na wynik modelu (średnia wartość $U$ w czasie $t=5.0$).
        \item \textbf{Analizowane modele:}
        \begin{itemize}
            \item Pełny model czasowo-przestrzenny (PDE).
            \item Uproszczony model czasowy (surogat ODE).
        \end{itemize}
        \item \textbf{Użyte metody:}
        \begin{itemize}
            \item \textbf{Metoda Morrisa:} Metoda przesiewowa, szybka, do identyfikacji najważniejszych parametrów. Mierzy ogólny wpływ ($\mu^*$) i interakcje/nieliniowość ($\sigma$).
            \item \textbf{Metoda Sobola:} Metoda globalna oparta na wariancji. Dokładniejsza, ale kosztowniejsza. Rozdziela wariancję wyjścia na wpływ poszczególnych parametrów ($S1$) i ich interakcje ($ST$).
        \end{itemize}
    \end{itemize}
\end{frame}

    \end{itemize}

    \begin{itemize}
        \item \textbf{Potwierdzenie:} Analiza Sobola potwierdza wyniki z metody Morrisa.
        \item \textbf{Wpływ całkowity (ST):} Parametr \textbf{q} ma dominujący wpływ.
        \item \textbf{Interakcje:} Duża różnica między $ST$ a $S1$ dla parametru $q$ wskazuje na jego silne interakcje z innymi parametrami.
        \item \textbf{Wpływ D:} Wpływ parametru $D$ na średnią wartość $U(t)$ jest marginalny.
    \end{itemize}
\end{frame}

\begin{frame}{Wnioski i Uproszczenie Modelu}
    \textbf{Wnioski z Analizy Wrażliwości:}
    \begin{itemize}
        \item Parametr imitacji ($q$) jest kluczowym czynnikiem sterującym dynamiką średniego poziomu adaptacji $U(t)$.
        \item Parametr innowacji ($p$) jest drugorzędny, ale wciąż istotny.
        \item Parametr dyfuzji ($D$) ma znikomy wpływ na \textit{średnią} wartość w systemie, chociaż jest kluczowy dla dynamiki \textit{przestrzennej}.
    \end{itemize}
    \vspace{1cm}
    \textbf{Propozycja uproszczenia modelu:}
    \begin{itemize}
        \item Model ODE jest już uproszczeniem, które ignoruje $D$. Analiza potwierdza, że jest to uzasadnione, jeśli interesuje nas tylko średnia wartość $U(t)$.
        \item W modelu PDE, parametr $D$ można by ustalić na stałą wartość (np. jego średnią z badanego zakresu) bez dużej utraty dokładności dla tej konkretnej miary wyjściowej. To redukuje złożoność problemu o jeden wymiar.
    \end{itemize}
\end{frame}

\section*{Zadanie 5 — Asymilacja danych (ABC i 3D-Var)}

\begin{frame}{Sformułowanie problemu asymilacji}
\begin{itemize}
  \item \textbf{Cel:} Odtworzenie parametrów modelu i stanu układu na podstawie zaszumionych obserwacji w oknie czasowym.
  \item \textbf{Metody:}
    \begin{itemize}
      \item \textbf{ABC (Approximate Bayesian Computation):} Metoda bezgradientowa, polegająca na losowaniu parametrów i akceptacji tych, które generują wyniki bliskie obserwacjom. Odporna na lokalne minima, ale kosztowna obliczeniowo.
      \item \textbf{3D-Var (Variational):} Minimalizacja funkcji kosztu $J(x) = ||y - H(x)||^2$. Szybka zbieżność przy dostępnych gradientach, ale ryzyko utknięcia w minimach lokalnych.
    \end{itemize}
  \item \textbf{Eksperyment:} Dane "rzeczywiste" z modelu PDE + szum Gaussowski. Testy dla modelu surogatowego ODE i pełnego PDE przy trzech budżetach czasowych (Low: 2s, Medium: 5s, High: 15s).
\end{itemize}
\end{frame}

\begin{frame}{Wyniki asymilacji dla modelu ODE}
\begin{columns}
\column{0.5\textwidth}
\begin{itemize}
  \item \textbf{3D-Var:} Bardzo szybka zbieżność (czas < 0.1s). Błąd RMSE minimalny ($\approx 0.006$). Metoda idealna dla prostego modelu ODE.
  \item \textbf{ABC:} Wymaga wielu iteracji (tysiące). Błąd stabilny ($\approx 0.012$), ale wyższy niż w 3D-Var.
  \item \textbf{Wnioski:} Dla tanich modeli numerycznych metody gradientowe (lub quasi-newtonowskie) są bezkonkurencyjne.
\end{itemize}

\section*{Zadanie 6 — Model PINN (Physics-Informed Neural Network)}

\begin{frame}{Koncepcja PINN}
\begin{itemize}
  \item \textbf{Idea:} Połączenie danych obserwacyjnych z wiedzą fizyczną zawartą w równaniach różniczkowych (PDE).
  \item \textbf{Architektura:} Sieć neuronowa $u_{NN}(x,t)$ aproksymująca rozwiązanie.
  \item \textbf{Funkcja kosztu:} $L = L_{data} + L_{physics}$
    \begin{itemize}
      \item $L_{data} = \frac{1}{N_d} \sum (u_{NN}(x_i, t_i) - u_{obs})^2$ (dopasowanie do danych)
      \item $L_{physics} = \frac{1}{N_f} \sum |f(u_{NN}(x_j, t_j))|^2$ (residuum PDE)
    \end{itemize}
  \item \textbf{Zaleta:} Możliwość trenowania na rzadkich danych i "uzupełniania" brakującej informacji fizyką.
\end{itemize}
\end{frame}

\begin{frame}{Implementacja i Przebieg Treningu}
\begin{itemize}
  \item \textbf{Sieć:} MLP (4 warstwy ukryte po 40 neuronów, aktywacja Tanh).
  \item \textbf{Dane:} 2000 losowych punktów z symulacji PDE (zasymilowane dane).
  \item \textbf{Kolokacja:} 10000 punktów do obliczania residuum PDE.
  \item \textbf{Czas treningu:} 62.63s (5000 epok).
  \item \textbf{Końcowa strata (Loss):} 0.000044 (Data: 0.000040, Phy: 0.000003).
\end{itemize}
